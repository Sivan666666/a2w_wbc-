cmake_minimum_required(VERSION 3.16)
project(a2w_z1_ppo10k LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(BODY_BRIDGE_ONLY "Build PC2 body bridge without Python or Z1 dependencies" OFF)
set(UNITREE_SDK2_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/unitree_sdk2" CACHE PATH "SDK2 checkout")
set(Z1_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/z1_sdk" CACHE PATH "Z1 SDK checkout")
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${UNITREE_SDK2_ROOT}" "sdk2")
add_executable(pc2_body_bridge cpp/pc2_body_bridge.cpp)
target_link_libraries(pc2_body_bridge PRIVATE unitree_sdk2 pthread)
add_executable(body_link_integration tests/body_link_integration.cpp)
target_include_directories(body_link_integration PRIVATE cpp)
target_link_libraries(body_link_integration PRIVATE unitree_sdk2 pthread)
if(BODY_BRIDGE_ONLY)
    return()
endif()
find_package(Python COMPONENTS Interpreter Development REQUIRED)
set(PYBIND11_FINDPYTHON ON)
find_package(pybind11 CONFIG REQUIRED)
find_package(Eigen3 REQUIRED)
find_library(Z1_LIBRARY NAMES Z1_SDK_${CMAKE_SYSTEM_PROCESSOR} PATHS "${Z1_SDK_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
add_executable(z1_arm_cycle cpp/z1_arm_cycle.cpp)
target_include_directories(z1_arm_cycle PRIVATE "${Z1_SDK_ROOT}/include")
target_link_libraries(z1_arm_cycle PRIVATE Eigen3::Eigen "${Z1_LIBRARY}" pthread)
set_target_properties(z1_arm_cycle PROPERTIES BUILD_RPATH "${Z1_SDK_ROOT}/lib")
pybind11_add_module(_hardware cpp/hardware.cpp)
target_include_directories(_hardware PRIVATE "${Z1_SDK_ROOT}/include")
target_link_libraries(_hardware PRIVATE unitree_sdk2 Eigen3::Eigen "${Z1_LIBRARY}" pthread)
set_target_properties(_hardware PROPERTIES
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/a2w_deploy"
    BUILD_RPATH "${Z1_SDK_ROOT}/lib;${UNITREE_SDK2_ROOT}/thirdparty/lib/${CMAKE_SYSTEM_PROCESSOR}")
